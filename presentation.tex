\begin{frame}{The paper and the agent's problem}
\small
\begin{columns}[T,totalwidth=\textwidth]
\begin{column}{0.39\textwidth}
  \key{One decision needs two predictions}
  \begin{itemize}\setlength\itemsep{0.06cm}
    \item \(X\): precision about the \textbf{common} state
    \item \(Y\): precision about this \textbf{particular} case
  \end{itemize}
  Their joint usefulness makes them complements.

  \vspace{0.10cm}
  \key{Knowledge collapse}\\[-0.02cm]
  Less effort weakens public learning; inherited \(X_t\) can converge to zero.
\end{column}
\begin{column}{0.58\textwidth}
  \key{Choose \(e\geq0\); take \(X,\tau_A\) as given}
  \[
  \max_{e\geq0}\;
  f(0,0)+G(X)\Delta_G+G(X)G(Y)\Delta_X-c(e)
  \]
  {\scriptsize
  \(G(X)\Delta_G\): probability \(\times\) stand-alone gain\\
  \(G(X)G(Y)\Delta_X\): joint probability \(\times\) complementarity gain\\
  \(c(e)\): resource cost of the chosen effort
  }

  \vspace{0.04cm}
  \[
  Y=\underbrace{\sigma^{-2}}_{\text{prior}}
  +\underbrace{\lambda_Ie}_{\text{human signal}}
  +\underbrace{\tau_A}_{\text{AI signal}}
  \]
  {\small Independent Gaussian evidence \(\Rightarrow\) precisions add.}
\end{column}
\end{columns}

\vfill
{\color{Coral}\rule{\textwidth}{0.8pt}}\\[-0.04cm]
{\small\textbf{Externality.} Effort also creates public precision
\(\lambda_GE_t\), but a short-lived atomistic agent does not value it.}
\end{frame}

\begin{frame}{Main result: complements and substitutes}
\small
\begin{columns}[T,totalwidth=\textwidth]
\begin{column}{0.43\textwidth}
  \key{Economic logic}

  \vspace{0.08cm}
  General knowledge makes case learning more valuable:
  \[
  X\uparrow\quad\Rightarrow\quad MB_e\uparrow.
  \]

  Effort and AI improve the same \(Y\); diminishing returns make them substitutes:
  \[
  \tau_A\uparrow\quad\Rightarrow\quad MB_e\downarrow.
  \]
\end{column}
\begin{column}{0.54\textwidth}
  \[
  U_e=\lambda_I\Delta_XG(X)g(Y)-e^{1/\varepsilon}=0
  \]
  \[
  \underbrace{U_{eX}}_{\text{complementarity}}
  =\lambda_I\Delta_Xg(X)g(Y)>0
  \]
  \[
  \underbrace{U_{e\tau_A}}_{\text{substitution}}
  =\lambda_I\Delta_XG(X)g'(Y)<0
  \]
  \key{Strict-sign domain}\\[-0.05cm]
  \(X,Y,\Delta_X,\lambda_I,\varepsilon>0\), with \(g>0\) and \(g'<0\).
\end{column}
\end{columns}

\vfill
\caveat{Boundary.} At \(X=0\), \(U_{e\tau_A}=0\) and \(g(0)\) is not finite.
Strict signs are interior; discrete comparisons preserve the weak global logic.
\end{frame}

\begin{frame}{What I did: test the boundary assumption}
\small
\begin{columns}[T,totalwidth=\textwidth]
\begin{column}{0.43\textwidth}
  \key{1. Audited Observation 1}\\[-0.01cm]
  The mechanism is right; strict cross-partials require \(X>0\).

  \vspace{0.25cm}
  \key{2. Relaxed \(\Delta_I=0\)}\\[-0.02cm]
  Baseline: case-specific correctness has \emph{no} value if the common
  prediction is wrong.

  \vspace{0.10cm}
  Useful benchmark, but strong: case knowledge may still improve triage when
  general knowledge is poor.
\end{column}
\begin{column}{0.54\textwidth}
  \key{Own extension: \(\Delta_I>0\)}
  {\small
  \[
  U_e=\lambda_Ig(Y)
  \underbrace{[\Delta_I+G(X)\Delta_X]}_{\substack{\text{stand-alone return}\\+\ \text{weighted complementarity}}}
  -e^{1/\varepsilon}.
  \]
  }
  At \(X=0\), the \(\Delta_I\) return survives:
  \[
  \Delta_I>0
  \Rightarrow e_\Delta(0,\tau_A)>0
  \Rightarrow F_\Delta(0)>0.
  \]

  \caveat{What this proves}\\[-0.02cm]
  Zero is not a fixed point. Positive low-knowledge states and welfare remain open.
\end{column}
\end{columns}

\vfill
{\scriptsize
\textbf{Lean audit:} derivatives, signs, boundary results, and the conditional
extension implication compile in Lean/Mathlib with no proof holes.\\[-0.02cm]
Section 5 changes aggregation, synthetic data, or public learning
\(e^\beta\); it does not relax \(\Delta_I=0\).}
\end{frame}
